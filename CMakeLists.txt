cmake_minimum_required(VERSION 3.24)
project(micro_gemma LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(MICRO_GEMMA_SANITIZERS "Enable Clang address/undefined behavior checks" OFF)
if(MICRO_GEMMA_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "The sanitizer preset currently requires Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer
        -fsanitize-ignorelist=${CMAKE_CURRENT_SOURCE_DIR}/tests/sanitizer-ignorelist.txt)
    add_link_options(-fsanitize=address,undefined)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option METAL CUDA BLAS OPENMP CPU_REPACK)
    set(GGML_${option} OFF CACHE BOOL "" FORCE)
endforeach()
# Only the ggml subtree is compiled. No llama model, tokenizer or inference code is linked.
FetchContent_Declare(ggml_source
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/b39a7bf1b038e8d1c3d58a313228dfc4f3993f5c
    URL_HASH SHA256=b95b2f88259483caeeb535c5667cf291ba3425ae85faea4872f440936eec14ad
    SOURCE_SUBDIR ggml)
FetchContent_MakeAvailable(ggml_source)
add_library(engine src/loader.cpp src/tokenizer.cpp src/operations.cpp src/model.cpp src/sampler.cpp src/profiler.cpp src/engine.cpp)
target_include_directories(engine PUBLIC src)
target_link_libraries(engine PUBLIC ggml ggml-cpu)
target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(inference_engine src/main.cpp)
target_link_libraries(inference_engine PRIVATE engine)
set_target_properties(inference_engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(unit_tests tests/unit.cpp)
    target_link_libraries(unit_tests PRIVATE engine)
    add_test(NAME unit COMMAND unit_tests)
    set(MICRO_GEMMA_MODEL "" CACHE FILEPATH "GGUF path for optional real-model integration tests")
    if(MICRO_GEMMA_MODEL)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:inference_engine> ${MICRO_GEMMA_MODEL})
        set_tests_properties(integration PROPERTIES TIMEOUT 600)
    endif()
endif()
